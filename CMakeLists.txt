function(add_build_core port id name)
  string(REGEX MATCH "${port}_game_objs :=[^\n]*\n([^\n]*\\\\\n)*[^\n]*" game_block "${upstream_make}")
  string(REGEX MATCHALL "[a-zA-Z0-9_]+\\.cpp" game_names "${game_block}")
  set(game_sources)
  foreach(file IN LISTS game_names)
    list(APPEND game_sources "source/${port}/src/${file}")
  endforeach()
  if(port STREQUAL "duke3d" OR port STREQUAL "sw")
    file(GLOB game_sources CONFIGURE_DEPENDS source/${port}/src/*.cpp)
    if(port STREQUAL "duke3d")
      list(FILTER game_sources EXCLUDE REGEX "/(in_android|m32.*|mdump|astub|sounds_mapster32|start.*)\\.cpp$")
    else()
      list(FILTER game_sources EXCLUDE REGEX "/(bldscript|brooms|jbhlp|jnstub|start.*)\\.cpp$")
    endif()
  endif()
  string(TOUPPER "${port}" upper_port)
  if(port STREQUAL "rr")
    file(GLOB n64 CONFIGURE_DEPENDS source/rr/src/n64/*.cpp)
    list(APPEND game_sources ${n64})
  endif()
  add_library(${id}_libretro SHARED $<TARGET_OBJECTS:build_libretro_engine> ${game_sources}
    libretro/core.cpp libretro/platform.cpp libretro/audio.cpp libretro/game_adapter.cpp)
  target_include_directories(${id}_libretro PRIVATE ${core_includes} source/${port}/src)
  target_compile_definitions(${id}_libretro PRIVATE LIBRETRO LR_GAME_${upper_port} NOASM NETCODE_DISABLE NDEBUG HAVE_VORBIS
    APPNAME="${name}" APPBASENAME="${id}" LIBRETRO_CORE_ID="${id}" LIBRETRO_GAME_NAME="${name}" LOGURU_WITH_STREAMS=0)
  if(MSVC)
    target_compile_options(${id}_libretro PRIVATE /W0 /bigobj)
  else()
    target_compile_options(${id}_libretro PRIVATE -w -funsigned-char -fno-strict-aliasing -fwrapv)
  endif()
  target_link_libraries(${id}_libretro PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  if(WIN32)
    target_link_libraries(${id}_libretro PRIVATE winmm ws2_32 shlwapi comctl32 ole32 uuid)
    if(MINGW)
      target_link_options(${id}_libretro PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  else()
    target_link_libraries(${id}_libretro PRIVATE m)
  endif()
  set_target_properties(${id}_libretro PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden)
  configure_file("libretro/info/${id}_libretro.info" "${id}_libretro.info" COPYONLY)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${id}.pk3")
    add_custom_command(TARGET ${id}_libretro POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/${id}.pk3" "$<TARGET_FILE_DIR:${id}_libretro>/${id}.pk3")
  endif()
endfunction()
set(LIBRETRO_GAMES "blood;duke3d;sw;rr;rrra;nam;napalm;ww2gi;exhumed;tekwar;witchaven;kenbuild" CACHE STRING "Game targets to compile")
foreach(port IN LISTS LIBRETRO_GAMES)
  if(port STREQUAL "blood")
    add_build_core(blood nblood "NBlood")
  elseif(port STREQUAL "duke3d")
    add_build_core(duke3d eduke32 "EDuke32")
  elseif(port STREQUAL "sw")
    add_build_core(sw voidsw "VoidSW")
  elseif(port STREQUAL "rr")
    add_build_core(rr rednukem "Rednukem")
  elseif(port STREQUAL "rrra" OR port STREQUAL "nam" OR port STREQUAL "napalm" OR port STREQUAL "ww2gi")
    add_build_core(rr ${port} "Rednukem ${port}")
    string(TOUPPER "${port}" variant)
    target_compile_definitions(${port}_libretro PRIVATE LR_VARIANT_${variant})
  elseif(port STREQUAL "exhumed")
    add_build_core(exhumed pcexhumed "PCExhumed")
  elseif(port STREQUAL "tekwar")
    add_build_core(tekwar etekwar "ETekWar")
  elseif(port STREQUAL "witchaven")
    add_build_core(witchaven ewitchaven "EWitchaven")
  elseif(port STREQUAL "kenbuild")
    add_build_core(kenbuild ekenbuild "EKenBuild")
  else()
    message(FATAL_ERROR "Unknown libretro game: ${port}")
  endif()
endforeach()
